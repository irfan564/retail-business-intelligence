Q23 Month-over-Month Revenue Growth (LAG) 
WITH monthly_rev AS (
    SELECT
        order_year, order_month, order_month_name,
        ROUND(SUM(order_value), 2) AS monthly_revenue
    FROM orders
    WHERE order_status = 'delivered'
    GROUP BY order_year, order_month, order_month_name
)
SELECT
    order_year, order_month, monthly_revenue,
    LAG(monthly_revenue) OVER (ORDER BY order_year, order_month) AS prev_month,
    ROUND(
        100.0 * (monthly_revenue - LAG(monthly_revenue) OVER (ORDER BY order_year, order_month))
        / NULLIF(LAG(monthly_revenue) OVER (ORDER BY order_year, order_month), 0), 2
    ) AS mom_growth_pct
FROM monthly_rev
ORDER BY order_year, order_month;
Q25 Top Sellers Ranked Within Each State (RANK PARTITION BY) 
WITH seller_rev AS (
    SELECT
        s.seller_state, s.seller_id,
        ROUND(SUM(oi.price), 2) AS total_revenue,
        COUNT(DISTINCT oi.order_id) AS total_orders
    FROM sellers s
    JOIN order_items oi ON s.seller_id = oi.seller_id
    GROUP BY s.seller_state, s.seller_id
)
SELECT
    seller_state, seller_id, total_revenue, total_orders,
    RANK() OVER (
        PARTITION BY seller_state
        ORDER BY total_revenue DESC
    ) AS rank_in_state
FROM seller_rev
ORDER BY seller_state, rank_in_state;
Q26 Customer RFM Segmentation Matrix (NTILE) 
WITH max_date AS (
    SELECT MAX(order_purchase_timestamp) AS max_timestamp FROM orders
),
rfm AS (
    SELECT
        c.customer_unique_id,
        DATE_PART('day', (SELECT max_timestamp FROM max_date) - MAX(o.order_purchase_timestamp)) AS recency,
        COUNT(DISTINCT o.order_id)  AS frequency,
        ROUND(SUM(o.order_value), 2) AS monetary
    FROM customers c
    JOIN orders o ON c.customer_id = o.customer_id
    WHERE o.order_status = 'delivered'
    GROUP BY c.customer_unique_id
)
SELECT
    customer_unique_id, recency, frequency, monetary,
    NTILE(4) OVER (ORDER BY recency  DESC) AS r_score,
    NTILE(4) OVER (ORDER BY frequency ASC)  AS f_score,
    NTILE(4) OVER (ORDER BY monetary  ASC)  AS m_score
FROM rfm LIMIT 100;
Q27 Customer Cohort Analysis (Monthly Acquisition & Retention) 
WITH customer_cohorts AS (
    SELECT
        c.customer_unique_id,
        MIN(DATE_TRUNC('month', o.order_purchase_timestamp)) AS cohort_month
    FROM customers c
    JOIN orders o ON c.customer_id = o.customer_id
    GROUP BY c.customer_unique_id
),
user_activities AS (
    SELECT
        c.customer_unique_id,
        DATE_TRUNC('month', o.order_purchase_timestamp) AS activity_month
    FROM customers c
    JOIN orders o ON c.customer_id = o.customer_id
)
SELECT
    cohort_month, activity_month,
    COUNT(DISTINCT ua.customer_unique_id) AS active_customers
FROM customer_cohorts cc
JOIN user_activities ua ON cc.customer_unique_id = ua.customer_unique_id
GROUP BY cohort_month, activity_month
ORDER BY cohort_month, activity_month;

Total Revenue = SUM( Orders[order_value] )

Average Order Value =
DIVIDE( [Total Revenue], [Total Orders], 0 )
📅 Time Intelligence 3 measures
YoY Revenue Growth %
YoY Revenue Growth Pct =
VAR CurrentRevenue = [Total Revenue]
VAR PreviousYearRevenue =
    CALCULATE(
        [Total Revenue],
        SAMEPERIODLASTYEAR( Dim_Calendar[Date] )
    )
RETURN
DIVIDE( CurrentRevenue - PreviousYearRevenue,
        PreviousYearRevenue, 0 )
YTD Cumulative Revenue
Cumulative Revenue YTD =
TOTALYTD( [Total Revenue], Dim_Calendar[Date] )
🚚 Logistics & Fulfillment 5 measures
Late Delivery %
Late Delivery Pct =
DIVIDE( [Late Orders Count], [Delivered Orders], 0 )
Fulfillment Rate %
Fulfillment Rate Pct =
DIVIDE( [Delivered Orders], [Total Orders], 0 )
⭐ Customer Satisfaction 3 measures
1-Star Review %
1 Star Review Pct =
DIVIDE(
    CALCULATE( COUNTROWS(Reviews),
               Reviews[review_score] = 1 ),
    COUNTROWS( Reviews ), 0
)
🎯 Advanced Analytics 2 measures
Repeat Customer Rate
Repeat Customer Rate =
VAR TotalUniqueCustomers =
    DISTINCTCOUNT( Orders[customer_id] )
VAR RepeatCustomers =
    COUNTROWS(
        FILTER(
            VALUES( Customers[customer_unique_id] ),
            CALCULATE( COUNTROWS( Orders ) ) > 1
        )
    )
RETURN
DIVIDE( RepeatCustomers, TotalUniqueCustomers, 0 )
Pareto Category Revenue Share
Top Categories Rev Share =
VAR TotalSalesAllCategories =
    CALCULATE(
        [Total Revenue],
        ALL( Products[product_category_name_english] )
    )
RETURN
DIVIDE( [Total Revenue], TotalSalesAllCategories, 0 )
